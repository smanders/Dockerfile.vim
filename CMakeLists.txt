cmake_minimum_required(VERSION 3.31)
if(NOT DEFINED CMAKE_INSTALL_PREFIX)
  # NOTE: CMAKE_INSTALL_PREFIX must be set before project() to take effect
  set(CMAKE_INSTALL_PREFIX ${CMAKE_BINARY_DIR}/dist CACHE PATH "Install path prefix, prepended onto install directories.")
endif()
project(Dockerfile.vim VERSION 21.09.06) # date-of-master-branch-commit
set(_dirs ftdetect ftplugin indent snippets syntax)
install(DIRECTORY ${_dirs} DESTINATION .)
set(CPACK_GENERATOR RPM)
set(CPACK_RPM_PACKAGE_ARCHITECTURE "noarch")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr/share/vim/vimfiles")
set(_cpack_package_version "v${PROJECT_VERSION}")
set(_release_tag_file "${CMAKE_CURRENT_SOURCE_DIR}/.github/release-tag.json")
if(EXISTS "${_release_tag_file}")
  file(READ "${_release_tag_file}" _release_tag_contents)
  string(JSON _release_tag ERROR_VARIABLE _release_tag_err GET "${_release_tag_contents}" tag)
  if(NOT _release_tag_err)
    string(STRIP "${_release_tag}" _release_tag)
    if(NOT _release_tag STREQUAL "")
      set(_cpack_package_version "${_release_tag}")
    endif()
  endif()
endif()
set(CPACK_PACKAGE_VERSION "${_cpack_package_version}")
set(CPACK_PACKAGE_FILE_NAME ${PROJECT_NAME}-${_cpack_package_version})
include(CPack)
